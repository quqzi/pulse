# WS2.1a — Schedule Touch Drag Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make moving and resizing existing schedule blocks work on touch devices (phone/iPad), not just mouse.

**Architecture:** Introduce a reusable `draggable` Svelte action built on **Pointer Events** (one code path for mouse + touch + pen), backed by a pure, unit-tested gesture reducer. `CalBlock` attaches the action to its body (move) and resize handle (resize); the action emits `start` / `move {dy}` / `end` which `CalBlock` re-dispatches to `WeekGrid`. `WeekGrid` drops its mouse-only `document` listeners for the move/resize paths and updates the drag ghost from the emitted deltas. The block-**create** path (long-press / mouse-drag on empty grid) is unchanged — it already works on touch.

**Tech Stack:** SvelteKit 2, TypeScript, Svelte actions, Pointer Events API, Vitest.

**Spec:** `docs/superpowers/specs/2026-06-02-ws2-multi-device-design.md` §5.3, §7.1. This is the first slice of WS2.1; follow-on plans cover anchored positioning + audit (§7.2) and tab-bar a11y + mobile audit (§7.3–7.4).

## Background (read before starting)

The bug: `CalBlock` (`src/lib/components/calendar/CalBlock.svelte`) starts drags on `on:mousedown` and types its events `MouseEvent` (`:15`, `:16`, `:61`, `:67`, `:85`, `:106`). `WeekGrid` (`src/lib/components/calendar/WeekGrid.svelte`) does the drag math in `onMousemove`/`onMouseup` attached to `document` on mouse events (`:369`, `:382`, `:323-324`, `:344-345`, `:365-366`). Touch never fires any of this, so **on a phone/iPad you cannot move or resize a block**.

Key existing facts to preserve:
- Move is **vertical-only, same day** — `onMousemove`'s `move` branch changes only `ghostTop` (`:375-376`); the day does not change mid-move. So a single `dy` delta is sufficient.
- Resize is **vertical-only** — changes only `ghostHeight` (`:377-378`).
- `snap`, `TOTAL_HEIGHT`, `START_HOUR`, `toLocalISO`, `weekDays`, `slotDayIdx`, `dragMode`, `dragSlot`, `dragSlotOrigTop`, `dragSlotOrigHeight`, `ghostTop`, `ghostHeight`, `ghostDayIdx`, `ghostVisible` already exist in `WeekGrid`.
- Tapping a block toggles its popover via `on:click` → `onBlockClick` (`:56-59`, `:84`). A drag must **not** also fire that click.

```ts
// src/lib/actions/dragGesture.ts
export interface Point {
  x: number;
  y: number;
}

export interface GestureState {
  origin: Point;
  started: boolean;
  /** true once a touch long-press has "grabbed"; mouse leaves this false and relies on threshold */
  grabbed: boolean;
}

export interface GestureConfig {
  /** px of movement before a (non-grabbed) drag starts */
  threshold: number;
}

export interface GestureResult {
  state: GestureState;
  emit: 'start' | 'move' | null;
  dx: number;
  dy: number;
}

export function createGesture(origin: Point, grabbed: boolean): GestureState {
  return { origin, started: false, grabbed };
}

export function moveGesture(state: GestureState, point: Point, cfg: GestureConfig): GestureResult {
  const dx = point.x - state.origin.x;
  const dy = point.y - state.origin.y;

  if (state.started) {
    return { state, emit: 'move', dx, dy };
  }
  const past = state.grabbed || Math.hypot(dx, dy) >= cfg.threshold;
  if (past) {
    return { state: { ...state, started: true }, emit: 'start', dx, dy };
  }
  return { state, emit: null, dx, dy };
}
```

## Task 2: `draggable` Svelte action

**Files:**
- Create: `src/lib/actions/draggable.ts`

This wraps the reducer with the DOM/Pointer-Events plumbing. On a **mouse** pointer it starts after the movement threshold; on **touch/pen** it requires a ~250ms **long-press** to "grab" (so taps open the popover and the gesture does not fight scrolling). It uses `setPointerCapture` so move events keep arriving, sets `touch-action: none` on the node so the browser does not pan once a drag is possible, and reports whether a real drag happened so the consumer can suppress the trailing click.

- [ ] **Step 1: Write the implementation**

```ts
// src/lib/actions/draggable.ts
import { createGesture, moveGesture, type GestureState, type Point } from './dragGesture';

export interface DraggableOptions {
  /** px of movement (mouse) before a drag begins. Default 4. */
  threshold?: number;
  /** ms a touch/pen must be held before a drag can begin. Default 250. */
  longPressMs?: number;
  onStart?: (e: PointerEvent) => void;
  onMove?: (detail: { dx: number; dy: number; event: PointerEvent }) => void;
  /** dragged = true if a real drag occurred (so the consumer can suppress the click). */
  onEnd?: (detail: { dragged: boolean; event: PointerEvent }) => void;
}

const DEFAULTS = { threshold: 4, longPressMs: 250 };

export function draggable(node: HTMLElement, options: DraggableOptions = {}) {
  let opts = { ...DEFAULTS, ...options };
  let state: GestureState | null = null;
  let activeId: number | null = null;
  let longPress: ReturnType<typeof setTimeout> | null = null;

  function clearLongPress() {
    if (longPress) { clearTimeout(longPress); longPress = null; }
  }

  function onPointerdown(e: PointerEvent) {
    if (activeId !== null) return;       // already tracking a pointer
    if (e.button > 0) return;            // primary button / touch only
    activeId = e.pointerId;
    state = createGesture({ x: e.clientX, y: e.clientY }, false);
    node.setPointerCapture(e.pointerId);
    node.addEventListener('pointermove', onPointermove);
    node.addEventListener('pointerup', onPointerup);
    node.addEventListener('pointercancel', onPointerup);

    if (e.pointerType !== 'mouse') {
      longPress = setTimeout(() => {
        longPress = null;
        if (state && !state.started) {
          state = { ...state, grabbed: true, started: true };
          opts.onStart?.(e);
        }
      }, opts.longPressMs);
    }
  }

  function onPointermove(e: PointerEvent) {
    if (state === null || e.pointerId !== activeId) return;
    const point: Point = { x: e.clientX, y: e.clientY };

    // Touch that moves before the long-press fires = the user is scrolling; abandon.
    if (!state.started && e.pointerType !== 'mouse' && !state.grabbed) {
      const moved = Math.hypot(point.x - state.origin.x, point.y - state.origin.y);
      if (moved > 10) { clearLongPress(); end(e, false); return; }
      return;
    }

    const res = moveGesture(state, point, { threshold: opts.threshold });
    state = res.state;
    if (res.emit === 'start') opts.onStart?.(e);
    if (res.emit !== null) {
      e.preventDefault();
      opts.onMove?.({ dx: res.dx, dy: res.dy, event: e });
    }
  }

  function onPointerup(e: PointerEvent) {
    if (e.pointerId !== activeId) return;
    clearLongPress();
    end(e, !!state?.started);
  }

  function end(e: PointerEvent, dragged: boolean) {
    opts.onEnd?.({ dragged, event: e });
    node.removeEventListener('pointermove', onPointermove);
    node.removeEventListener('pointerup', onPointerup);
    node.removeEventListener('pointercancel', onPointerup);
    try { node.releasePointerCapture(e.pointerId); } catch { /* already released */ }
    state = null;
    activeId = null;
  }

  node.addEventListener('pointerdown', onPointerdown);
  node.style.touchAction = 'none';

  return {
    update(next: DraggableOptions) { opts = { ...DEFAULTS, ...next }; },
    destroy() {
      clearLongPress();
      node.removeEventListener('pointerdown', onPointerdown);
    },
  };
}
```
