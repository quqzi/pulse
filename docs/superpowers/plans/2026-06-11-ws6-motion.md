# WS6 — Motion & Design Refinement Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give Pulse a documented motion language — tokens, list stagger, stat count-ups, refined press states — applied app-wide, with every animation honoring `prefers-reduced-motion`. Refinement, not redesign: the three design languages (Vercel/Apple/Resend) stay.

**Architecture:** Motion tokens (`--ease-*`, `--dur-*`) join the existing CSS custom-property token system in `app.css`; the 11 hardcoded `cubic-bezier`/duration sites are swapped to tokens. New motion utilities are additive: a `.stagger-children` CSS utility, a `countup` Svelte action with a pure, tested interpolator, and a `.pressable` press state. One global reduced-motion block guards everything.

**Tech Stack:** Plain CSS custom properties (no animation lib), Svelte 4 actions, Vitest for the pure interpolator. Skills: `css-animations`, `ui-ux-pro-max` for the final review.

```ts
import { describe, it, expect } from 'vitest';
import { countupFrames } from './countup';

describe('countupFrames', () => {
  it('starts at 0 and ends exactly at the target', () => {
    const frames = countupFrames(43, 5);
    expect(frames[0]).toBe(0);
    expect(frames[frames.length - 1]).toBe(43);
  });
  it('is monotonically non-decreasing with ease-out spacing', () => {
    const frames = countupFrames(100, 10);
    for (let i = 1; i < frames.length; i++) expect(frames[i]).toBeGreaterThanOrEqual(frames[i - 1]);
    // ease-out: first step covers more ground than the last
    expect(frames[1] - frames[0]).toBeGreaterThan(frames[frames.length - 1] - frames[frames.length - 2]);
  });
  it('handles decimals to one place', () => {
    const frames = countupFrames(3.7, 4, 1);
    expect(frames[frames.length - 1]).toBe(3.7);

});
```

```ts
// src/lib/motion/countup.ts
/** Frame values from 0 → target with ease-out spacing. Pure — testable. */
export function countupFrames(target: number, steps: number, decimals = 0): number[] {
  const f = Math.pow(10, decimals);
  const frames: number[] = [];
  for (let i = 0; i <= steps; i++) {
    const t = i / steps;
    const eased = 1 - Math.pow(1 - t, 3); // cubic ease-out
    frames.push(Math.round(target * eased * f) / f);
  }
  return frames;
}

/** Svelte action: animate the element's text from 0 to `value` on mount/update. */
export function countup(node: HTMLElement, opts: { value: number; decimals?: number; durationMs?: number }) {
  let raf = 0;
  const reduced = typeof matchMedia !== 'undefined' && matchMedia('(prefers-reduced-motion: reduce)').matches;

  function run(value: number, decimals = 0, durationMs = 600) {
    cancelAnimationFrame(raf);
    if (reduced || value === 0) { node.textContent = value.toFixed(decimals); return; }
    const frames = countupFrames(value, Math.max(1, Math.round(durationMs / 16)), decimals);
    let i = 0;
    const tick = () => {
      node.textContent = frames[i].toFixed(decimals);
      if (++i < frames.length) raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);
  }

  run(opts.value, opts.decimals, opts.durationMs);
  return {
    update(next: { value: number; decimals?: number; durationMs?: number }) {
      run(next.value, next.decimals, next.durationMs);
    },
    destroy() { cancelAnimationFrame(raf); },
  };
}
```
