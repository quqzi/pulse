# WS1.1 — Test Harness & Characterization Tests Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up Vitest and pin the current behaviour of `srs.ts` (SM-2) and `scheduler.ts` (auto-schedule) with characterization tests, so the WS1.2+ repository refactor has a safety net.

**Architecture:** Add a standalone `vitest.config.ts` (Node environment, `$lib` alias resolved manually so SvelteKit's Vite plugin isn't needed). Co-locate `*.test.ts` next to the modules under `src/lib/`. Tests are *characterization* tests — they assert the code's **current** behaviour and pass immediately against existing code; their job is to fail loudly if a later refactor changes that behaviour. Both modules call `new Date()` and `scheduler` calls `Math.random()`, so every test fakes time and stubs randomness for determinism.

**Tech Stack:** Vitest, TypeScript, SvelteKit 2 (Vite 8). No jsdom — these are headless logic tests.

**Source of truth:** `docs/superpowers/specs/2026-05-29-pulse-roadmap-and-foundation-design.md` §4.5.

Create `src/lib/srs.test.ts`:
```ts
import { describe, it, expect, beforeEach, afterEach, vi } from 'vitest';
import { reviewCard, defaultCard, isDue } from '$lib/srs';

// Fixed instant at noon UTC so the UTC date (used by toISOString) is stable
// regardless of the machine's timezone.
beforeEach(() => {
  vi.useFakeTimers();
  vi.setSystemTime(new Date('2026-01-15T12:00:00Z'));
});
afterEach(() => {
  vi.useRealTimers();
});

describe('reviewCard', () => {
  it('Again (1) resets interval to 1 and drops ease by 0.2', () => {
    const r = reviewCard(2.5, 10, 1);
    expect(r.intervalDays).toBe(1);
    expect(r.ease).toBeCloseTo(2.3);
    expect(r.dueDate).toBe('2026-01-16');
  });

  it('Hard (2) multiplies interval by 1.2 and drops ease by 0.15', () => {
    const r = reviewCard(2.5, 10, 2);
    expect(r.intervalDays).toBe(12); // round(10 * 1.2)
    expect(r.ease).toBeCloseTo(2.35);
    expect(r.dueDate).toBe('2026-01-27');
  });

  it('Good (3) multiplies interval by ease and leaves ease unchanged', () => {
    const r = reviewCard(2.5, 10, 3);
    expect(r.intervalDays).toBe(25); // round(10 * 2.5)
    expect(r.ease).toBeCloseTo(2.5);
    expect(r.dueDate).toBe('2026-02-09');
  });

  it('Easy (4) multiplies interval by ease*1.3 and raises ease by 0.1', () => {
    const r = reviewCard(2.0, 10, 4);
    expect(r.intervalDays).toBe(26); // round(10 * 2.0 * 1.3)
    expect(r.ease).toBeCloseTo(2.1);
    expect(r.dueDate).toBe('2026-02-10');
  });

  it('floors ease at 1.3 on repeated Again', () => {
    const r = reviewCard(1.3, 5, 1);
    expect(r.ease).toBeCloseTo(1.3);
    expect(r.intervalDays).toBe(1);
  });
});

describe('defaultCard', () => {
  it('returns a fresh card due today', () => {
    expect(defaultCard()).toEqual({
      ease: 2.5,
      intervalDays: 1,
      dueDate: '2026-01-15',
      reps: 0,
    });
  });
});

describe('isDue', () => {
  it('is true for past and today, false for future', () => {
    expect(isDue('2026-01-14')).toBe(true);
    expect(isDue('2026-01-15')).toBe(true);
    expect(isDue('2026-01-16')).toBe(false);
  });
});
```

### Task 3: Characterization tests for `scheduler.ts`

**Files:**
- Create: `src/lib/scheduler.test.ts`
- Under test: `src/lib/scheduler.ts` (unchanged)

Reference behaviour (`src/lib/scheduler.ts`): `generateSchedule(input)` returns AI-suggested `ScheduleSlot[]`. With no subjects and no tasks it returns `[]`. Subjects with an hours deficit get greedy 06:00-anchored blocks (neutral energy → earliest slot wins). Confirmed slots + school hours are blocked. Each unfinished task with a `subjectId` and in-week `dueDate` gets a 45-min block on/just before its due date. Slot ids are `'ai_' + Math.random()...`; all suggested slots have `isAiSuggested: true`, `isConfirmed: false`.

- [ ] **Step 1: Write the characterization tests**

Create `src/lib/scheduler.test.ts`:
```ts
import { describe, it, expect, beforeEach, afterEach, vi } from 'vitest';
import { generateSchedule, type SchedulerInput } from '$lib/scheduler';
import type { Settings } from '$lib/types';

// 2026-01-12 is a Monday. Parsed without 'Z' => local midnight, matching how the
// scheduler builds week days with local date methods.
const WEEK_START = new Date('2026-01-12T00:00:00');

function baseSettings(over: Partial<Settings> = {}): Settings {
  return {
    theme: 'vercel-light',
    lang: 'en',
    dailyGoalHours: 2,
    pomodoroWork: 25,
    pomodoroBreak: 5,
    schoolHoursEnabled: false,
    schoolDays: [1, 2, 3, 4, 5],
    schoolStart: '08:00',
    schoolEnd: '15:30',
    energyProfile: 'neutral',
    bufferMinutes: 10,
    // NOTE: `Settings` in $lib/types has no `examDate` yet (the settings.ts copy does —
    // the two are unified in WS1.4 / gap G10). The scheduler reads input.examDate, not
    // settings.examDate, so we intentionally omit it here to match the $lib/types shape.
    ...over,
  };
}

function baseInput(over: Partial<SchedulerInput> = {}): SchedulerInput {
  return {
    weekStart: WEEK_START,
    subjects: [],
    sessions: [],
    srsCards: [],
    tasks: [],
    existingSlots: [],
    settings: baseSettings(),
    examDate: null,
    ...over,
  };
}

// Minutes-from-midnight from a "YYYY-MM-DDTHH:MM:00" local-time string.
function startMin(iso: string): number {
  const [h, m] = iso.split('T')[1].split(':').map(Number);
  return h * 60 + m;
}

beforeEach(() => {
  vi.useFakeTimers();
  vi.setSystemTime(new Date('2026-01-12T08:00:00Z'));
  vi.spyOn(Math, 'random').mockReturnValue(0.5); // deterministic ids
});
afterEach(() => {
  vi.useRealTimers();
  vi.restoreAllMocks();
});

describe('generateSchedule', () => {
  it('returns nothing when there are no subjects and no tasks', () => {
    expect(generateSchedule(baseInput())).toEqual([]);
  });

  it('fills the earliest available 06:00 block for a subject with a deficit', () => {
    const out = generateSchedule(baseInput({
      subjects: [{ id: 's1', userId: 'u', name: 'Physics', symbol: 'Ph', color: '#0d9488', colorRgb: '13,148,136', orderIndex: 0 }],
    }));
    expect(out.length).toBeGreaterThan(0);
    expect(out.every(s => s.isAiSuggested && !s.isConfirmed && !s.isFocusBlock)).toBe(true);
    expect(out.every(s => s.subjectId === 's1')).toBe(true);
    expect(out.every(s => s.id.startsWith('ai_'))).toBe(true);
    expect(out[0].startTime).toBe('2026-01-12T06:00:00');
    expect(out[0].endTime).toBe('2026-01-12T07:30:00');
  });

  it('never schedules a suggested block overlapping school hours', () => {
    const out = generateSchedule(baseInput({
      subjects: [{ id: 's1', userId: 'u', name: 'Physics', symbol: 'Ph', color: '#0d9488', colorRgb: '13,148,136', orderIndex: 0 }],
      settings: baseSettings({ schoolHoursEnabled: true, schoolDays: [1], schoolStart: '08:00', schoolEnd: '15:30' }),
    }));
    const mondaySlots = out.filter(s => s.startTime.startsWith('2026-01-12'));
    for (const s of mondaySlots) {
      const start = startMin(s.startTime);
      const end = startMin(s.endTime);
      // must lie entirely before 08:00 (480) or after 15:30 (930)
      expect(end <= 480 || start >= 930).toBe(true);
    }
  });

  it('injects a 45-minute block on a task due date', () => {
    const out = generateSchedule(baseInput({
      tasks: [{
        id: 't1', userId: 'u', subjectId: 's1', title: 'Mock paper',
        priority: 'high', dueDate: '2026-01-14', createdAt: '2026-01-12T00:00:00.000Z',
      }],
    }));
    expect(out.length).toBe(1);
    expect(out[0].subjectId).toBe('s1');
    expect(out[0].startTime).toBe('2026-01-14T06:00:00');
    expect(out[0].endTime).toBe('2026-01-14T06:45:00');
    expect(out[0].isAiSuggested).toBe(true);
  });
});
```
