# WS1.2 — Repository Engine + Adapters + Toast Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the dependency-injected repository engine, its local + mock adapters, a toast feedback store/component, and an `id` helper — all unit-tested — without changing any existing app data flow.

**Architecture:** The repository core lives in a **pure `engine.ts`** whose dependencies (current auth mode, adapter selection, error reporting) are *injected*. This keeps the engine and its tests free of SvelteKit virtual modules (`$env`, `$app/environment`) that the real Supabase client and `authCtx` transitively import and that the Node Vitest config can't resolve. The engine implements optimistic **write-through with snapshot rollback** (a failed write reverts the store and toasts; the repo owns the failure and does not re-throw) and **read-empty-on-error** (a failed read leaves the store unchanged and never rejects) — the contracts from spec §4.1a (G8/G13). This task is **purely additive**: the Supabase adapter, production wiring (`repository.ts`), and per-entity configs land in WS1.3, so the live app keeps running on the existing `db.ts`.

**Tech Stack:** Svelte 5 stores, TypeScript, Vitest (from WS1.1).

**Source of truth:** `docs/superpowers/specs/2026-05-29-pulse-roadmap-and-foundation-design.md` §4.1, §4.1a, §4.4.

**Prerequisite:** WS1.1 merged or present (Vitest harness, `test:unit`/`check:all` scripts).

```ts
import { writable } from 'svelte/store';

export type ToastKind = 'error' | 'success' | 'info';
export interface Toast {
  id: string;
  kind: ToastKind;
  message: string;
}

export const toasts = writable<Toast[]>([]);

let seq = 0;

function push(kind: ToastKind, message: string, ttlMs = 4000): string {
  const id = `t${++seq}`;
  toasts.update((list) => [...list, { id, kind, message }]);
  if (ttlMs > 0) setTimeout(() => dismissToast(id), ttlMs);
  return id;
}

export function dismissToast(id: string): void {
  toasts.update((list) => list.filter((t) => t.id !== id));
}

export const toast = {
  error: (message: string) => push('error', message),
  success: (message: string) => push('success', message),
  info: (message: string) => push('info', message),
};
```

```ts
import { writable, get } from 'svelte/store';
import { createRepository } from '$lib/data/engine';
import { mockAdapter } from '$lib/data/adapters/mock';

type Row = { id: string; n: number };

function deps(adapter: ReturnType<typeof mockAdapter<Row>>, onError = () => {}) {
  return {
    getMode: () => 'guest' as const,
    selectAdapter: () => adapter,
    onError,
  };
}

describe('createRepository', () => {
  it('load sets the store from the adapter, sorted by the configured order', async () => {
    const store = writable<Row[]>([]);
    const adapter = mockAdapter<Row>([{ id: 'b', n: 2 }, { id: 'a', n: 1 }]);
    const repo = createRepository<Row>({ store, order: (x, y) => x.n - y.n }, deps(adapter));
    const out = await repo.load();
    expect(out.map((r) => r.id)).toEqual(['a', 'b']);
    expect(get(store).map((r) => r.id)).toEqual(['a', 'b']);
  });

  it('save upserts into the store and persists via the adapter', async () => {
    const store = writable<Row[]>([{ id: 'a', n: 1 }]);
    const adapter = mockAdapter<Row>([{ id: 'a', n: 1 }]);
    const repo = createRepository<Row>({ store }, deps(adapter));
    await repo.save({ id: 'b', n: 2 });
    expect(get(store)).toContainEqual({ id: 'b', n: 2 });
    expect(adapter.data).toContainEqual({ id: 'b', n: 2 });
  });

  it('reverts the store and reports an error when a save fails', async () => {
    const store = writable<Row[]>([{ id: 'a', n: 1 }]);
    const adapter = mockAdapter<Row>([{ id: 'a', n: 1 }]);
    adapter.failSave = true;
    const onError = vi.fn();
    const repo = createRepository<Row>({ store }, deps(adapter, onError));
    await repo.save({ id: 'b', n: 2 });
    expect(get(store)).toEqual([{ id: 'a', n: 1 }]); // rolled back
    expect(onError).toHaveBeenCalledOnce();
  });

  it('remove deletes optimistically and reverts on failure', async () => {
    const store = writable<Row[]>([{ id: 'a', n: 1 }, { id: 'b', n: 2 }]);
    const adapter = mockAdapter<Row>([{ id: 'a', n: 1 }, { id: 'b', n: 2 }]);
    const onError = vi.fn();
    const repo = createRepository<Row>({ store }, deps(adapter, onError));
    await repo.remove('a');
    expect(get(store).map((r) => r.id)).toEqual(['b']);
    adapter.failRemove = true;
    await repo.remove('b');
    expect(get(store).map((r) => r.id)).toEqual(['b']); // reverted
    expect(onError).toHaveBeenCalledOnce();
  });

  it('leaves the store unchanged and does not throw on read error', async () => {
    const store = writable<Row[]>([{ id: 'a', n: 1 }]);
    const adapter = mockAdapter<Row>([]);
    adapter.failList = true;
    const repo = createRepository<Row>({ store }, deps(adapter));
    await expect(repo.load()).resolves.toEqual([{ id: 'a', n: 1 }]);

    const guest = mockAdapter<Row>([{ id: 'g', n: 1 }]);
    const auth = mockAdapter<Row>([{ id: 'u', n: 1 }]);

    const repo = createRepository<Row>(

    );
    await repo.load();
    expect(get(store).map((r) => r.id)).toEqual(['g']);
    mode = 'authenticated';
    await repo.load();
    expect(get(store).map((r) => r.id)).toEqual(['u']);
  });
});
```

```ts
import type { Adapter } from '$lib/data/engine';

export interface MockAdapter<T> extends Adapter<T> {
  data: T[];
  failList: boolean;
  failSave: boolean;
  failRemove: boolean;
}

/** In-memory adapter for unit tests. Toggle failList/failSave/failRemove to simulate errors. */
export function mockAdapter<T extends { id: string }>(seed: T[] = []): MockAdapter<T> {
  const a: MockAdapter<T> = {
    data: [...seed],
    failList: false,
    failSave: false,
    failRemove: false,
    async list() {
      if (a.failList) throw new Error('mock list error');
      return [...a.data];
    },
    async save(item) {
      if (a.failSave) throw new Error('mock save error');
      const i = a.data.findIndex((x) => x.id === item.id);
      if (i >= 0) a.data[i] = item;
      else a.data.push(item);
    },
    async remove(id) {
      if (a.failRemove) throw new Error('mock remove error');
      a.data = a.data.filter((x) => x.id !== id);
    },
  };
  return a;
}
```

```ts
import { get, type Writable } from 'svelte/store';

export type Mode = 'guest' | 'authenticated';

/** Storage backend for one entity. Implemented by local (WS1.2), supabase (WS1.3), mock (tests). */
export interface Adapter<T> {
  list(): Promise<T[]>;
  save(item: T): Promise<void>;
  remove(id: string): Promise<void>;
}

export interface RepoConfig<T extends { id: string }> {
  store: Writable<T[]>;
  /** Optional client-side sort applied after every load and write-through. */
  order?: (a: T, b: T) => number;
}

/** Injected so the engine (and its tests) never import $env/$app via supabase/authCtx. */
export interface RepoDeps<T extends { id: string }> {
  getMode(): Mode;
  selectAdapter(mode: Mode): Adapter<T>;
  onError(message: string): void;
}

export interface Repository<T extends { id: string }> {
  load(): Promise<T[]>;
  save(item: T): Promise<void>;
  remove(id: string): Promise<void>;
}

export function createRepository<T extends { id: string }>(
  cfg: RepoConfig<T>,
  deps: RepoDeps<T>,
): Repository<T> {
  const sort = (items: T[]): T[] => (cfg.order ? [...items].sort(cfg.order) : items);
  const adapter = (): Adapter<T> => deps.selectAdapter(deps.getMode());

  function upsert(arr: T[], item: T): T[] {
    const i = arr.findIndex((x) => x.id === item.id);
    const next = i >= 0 ? arr.map((x) => (x.id === item.id ? item : x)) : [...arr, item];
    return sort(next);
  }

  return {
    // Read-empty-on-error (G13): never reject; leave the store at its prior value on failure.
    async load() {
      try {
        const items = sort(await adapter().list());
        cfg.store.set(items);
        return items;
      } catch {
        return get(cfg.store);
      }
    },

    // Optimistic write-through with rollback (G8): the repo owns failure (no re-throw).
    async save(item) {
      const snapshot = get(cfg.store);
      cfg.store.set(upsert(snapshot, item));
      try {
        await adapter().save(item);
      } catch {
        cfg.store.set(snapshot);
        deps.onError("Couldn't save — try again");
      }
    },

    async remove(id) {
      const snapshot = get(cfg.store);
      cfg.store.set(snapshot.filter((x) => x.id !== id));
      try {
        await adapter().remove(id);
      } catch {
        cfg.store.set(snapshot);
        deps.onError("Couldn't delete — try again");
      }
    },
  };
}
```

```ts
import type { Adapter } from '$lib/data/engine';

/**
 * localStorage-backed adapter. Stores a JSON array of camelCase entities under `lsKey`.
 * Uses a `typeof localStorage` guard instead of $app/environment so it stays importable in Node tests.
 */
export function localAdapter<T extends { id: string }>(lsKey: string): Adapter<T> {
  const read = (): T[] => {
    if (typeof localStorage === 'undefined') return [];
    try {
      return JSON.parse(localStorage.getItem(lsKey) ?? '[]') as T[];
    } catch {
      return [];
    }
  };
  const write = (items: T[]): void => {
    if (typeof localStorage === 'undefined') return;
    localStorage.setItem(lsKey, JSON.stringify(items));
  };

  return {
    async list() {
      return read();
    },
    async save(item) {
      const all = read();
      const i = all.findIndex((x) => x.id === item.id);
      if (i >= 0) all[i] = item;
      else all.push(item);
      write(all);
    },
    async remove(id) {
      write(read().filter((x) => x.id !== id));
    },
  };
}
```
