<script lang="ts">
  import type { StudySession, Subject } from '$lib/types';

  export let sessions: StudySession[] = [];
  export let subjects: Subject[] = [];

  const WEEKS = 16;
  const INTENSITY_HOURS = [0, 0.5, 1.5, 3]; // hours threshold for level 1 / 2 / 3 / 4

  // Build a map: YYYY-MM-DD → StudySession[]
  function sessionsByDay(ss: StudySession[]): Map<string, StudySession[]> {
    const map = new Map<string, StudySession[]>();
    for (const s of ss) {
      const day = s.startedAt.split('T')[0];
      if (!map.has(day)) map.set(day, []);
      map.get(day)!.push(s);
    }
    return map;
  }

  function formatDuration(sec: number): string {
    if (sec <= 0) return '0m';
    const h = Math.floor(sec / 3600);
    const m = Math.round((sec % 3600) / 60);
    if (h === 0) return `${m}m`;
    if (m === 0) return `${h}h`;
    return `${h}h ${m}m`;
  }

  // Returns intensity level 0-4 from hours value
  function intensityLevel(hours: number): number {
    if (hours <= 0) return 0;
    for (let i = INTENSITY_HOURS.length - 1; i >= 0; i--) {
      if (hours >= INTENSITY_HOURS[i]) return i + 1;
    }
    return 1;
  }

  // Build the grid: array of week-columns, each column = array of 7 day-cells (Mon–Sun)
  // gridData[col][row] = { dateStr, level, tooltip }
  type Cell = { dateStr: string; level: number; tooltip: string; inRange: boolean };

  $: byDay = sessionsByDay(sessions);

  $: gridData = (() => {
    // The grid ends on the Sunday of the current week (or today if today is Sunday)
    const today = new Date();
    // Find most-recent Sunday (end of last complete week column)
    const endSunday = new Date(today);
    // Advance to the coming Sunday (or stay if today IS Sunday)
    const dayOfWeek = today.getDay(); // 0=Sun
    endSunday.setDate(today.getDate() + (dayOfWeek === 0 ? 0 : 7 - dayOfWeek));
    endSunday.setHours(0, 0, 0, 0);

    const startDate = new Date(endSunday);
    // Go back WEEKS * 7 days to find Monday that starts the grid
    startDate.setDate(endSunday.getDate() - WEEKS * 7 + 1); // +1 so we get exactly WEEKS weeks

    const todayStr = today.toISOString().split('T')[0];
    const cols: Cell[][] = [];

    for (let w = 0; w < WEEKS; w++) {
      const col: Cell[] = [];
      for (let d = 0; d < 7; d++) {
        // d=0 is Monday, d=6 is Sunday
        const cell = new Date(startDate);
        cell.setDate(startDate.getDate() + w * 7 + d);
        const dateStr = cell.toISOString().split('T')[0];
        const isFuture = dateStr > todayStr;
        const daySessions = byDay.get(dateStr) ?? [];
        const totalSec = daySessions.reduce((s, x) => s + x.durationSec, 0);
        const hours = totalSec / 3600;
        const level = isFuture ? 0 : intensityLevel(hours);
        const dayLabel = cell.toLocaleDateString('en-US', { weekday: 'short', month: 'short', day: 'numeric' });
        const tooltip = isFuture
          ? ''
          : daySessions.length === 0
          ? `${dayLabel} · No study`
          : `${dayLabel} · ${formatDuration(totalSec)} · ${daySessions.length} session${daySessions.length === 1 ? '' : 's'}`;
        col.push({ dateStr, level, tooltip, inRange: !isFuture });
      }
      cols.push(col);
    }
    return cols;
  })();

  // Month labels: one per column, show month name when a new month starts within that week
  $: monthLabels = (() => {
    return gridData.map((col) => {
      // Use the Monday of this week column
      const mondayStr = col[0].dateStr;
      const mondayDate = new Date(mondayStr + 'T00:00:00');
      // Show label only when Monday is the 1st–7th of the month (first week of month)
      if (mondayDate.getDate() <= 7) {
        return mondayDate.toLocaleDateString('en-US', { month: 'short' });
      }
      return '';
    });
  })();
</script>

<script lang="ts">
  import { onMount } from 'svelte';
  import { auth } from '$lib/stores/auth';
  import { subjects, topics, sessions, streak, loadData } from '$lib/stores/data';
  import HeatmapGrid from '$lib/components/journey/HeatmapGrid.svelte';

  const TIMELINE_PAGE_SIZE = 30;

  // All-time total hours (all loaded sessions)
  $: totalHours = $sessions.reduce((sum, s) => sum + s.durationSec / 3600, 0);
  $: totalSessions = $sessions.length;

  function formatHours(h: number): string {
    if (h < 1) return `${Math.round(h * 60)}m`;
    const hh = Math.floor(h);
    const mm = Math.round((h - hh) * 60);
    return mm ? `${hh}h ${mm}m` : `${hh}h`;
  }

  function formatDuration(sec: number): string {
    if (sec <= 0) return '0m';
    const h = Math.floor(sec / 3600);
    const m = Math.round((sec % 3600) / 60);
    if (h === 0) return `${m}m`;
    if (m === 0) return `${h}h`;
    return `${h}h ${m}m`;
  }

  // Group sessions by YYYY-MM-DD, sorted newest first
  type DayGroup = {
    dateStr: string;      // YYYY-MM-DD
    label: string;        // "Monday, May 26"
    totalSec: number;
    sessions: typeof $sessions;
  };

  $: subjectMap = Object.fromEntries($subjects.map(s => [s.id, s]));
  $: topicMap = Object.fromEntries($topics.map(t => [t.id, t]));

  $: dayGroups = (() => {
    const map = new Map<string, typeof $sessions>();
    for (const s of $sessions) {
      const day = s.startedAt.split('T')[0];
      if (!map.has(day)) map.set(day, []);
      map.get(day)!.push(s);
    }
    // Sort days newest first
    const days = Array.from(map.entries()).sort((a, b) => b[0].localeCompare(a[0]));
    return days.map(([dateStr, daySessions]) => {
      const d = new Date(dateStr + 'T00:00:00');
      const label = d.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' });
      const totalSec = daySessions.reduce((sum, s) => sum + s.durationSec, 0);
      // Sort sessions within day newest first
      const sorted = [...daySessions].sort((a, b) => b.startedAt.localeCompare(a.startedAt));
      return { dateStr, label, totalSec, sessions: sorted };
    });
  })();

  $: visibleDayGroups = dayGroups.slice(0, TIMELINE_PAGE_SIZE);
  $: hasMore = dayGroups.length > TIMELINE_PAGE_SIZE;

  // Rough count: show "and N more days" when truncated
  $: remainingDays = dayGroups.length - TIMELINE_PAGE_SIZE;

  function formatTimeRange(startedAt: string, endedAt: string): string {
    const fmt = (iso: string) =>
      new Date(iso).toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', hour12: true });
    return `${fmt(startedAt)} – ${fmt(endedAt)}`;
  }

  const MODE_LABELS: Record<string, string> = {
    pomodoro: 'Pomodoro',
    free: 'Free',
    quick: 'Quick',
  };

  onMount(() => loadData($auth));
</script>
