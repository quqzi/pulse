# Scores Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a fully functional Scores page that lets users record mock exam scores per subject, view them filtered by subject, and see a trend bar chart inline per subject.

**Architecture:** A `MockScore` record stores a raw score, max possible score, date taken, and optional notes. The data layer follows the existing `db.ts` guest/authenticated dual-path pattern with `pulse_mock_scores` in localStorage and a `mock_scores` Supabase table. The page uses a local `activeSubject` filter variable and a separate `ScoreForm.svelte` component for the add/edit form.

**Tech Stack:** SvelteKit 2, TypeScript, Supabase JS v2, plain CSS tokens

Run this in the Supabase SQL editor (Dashboard → SQL Editor → New Query):

```sql
create table mock_scores (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  subject_id uuid not null references subjects(id) on delete cascade,
  score numeric not null,
  max_score numeric not null default 100,
  taken_at date not null,
  notes text,
  created_at timestamptz not null default now()
);

alter table mock_scores enable row level security;

create policy "Users own mock_scores"
  on mock_scores for all
  using (auth.uid() = user_id);

create index mock_scores_user_id_idx on mock_scores (user_id);
create index mock_scores_taken_at_idx on mock_scores (taken_at desc);
```

```ts
  // ── MOCK SCORES ───────────────────────────────────────────
  async getScores(userId: string, mode: Mode): Promise<MockScore[]> {
    if (mode === 'guest') return lsGet<MockScore>('pulse_mock_scores');
    const { data } = await supabase.from('mock_scores').select('*')
      .eq('user_id', userId)
      .order('taken_at', { ascending: false });
    return (data ?? []).map(mapScoreFromDb);
  },

  async saveScore(score: MockScore, mode: Mode): Promise<void> {
    if (mode === 'guest') {
      const all = lsGet<MockScore>('pulse_mock_scores');
      const idx = all.findIndex(s => s.id === score.id);
      if (idx >= 0) all[idx] = score; else all.unshift(score);
      lsSet('pulse_mock_scores', all);
      return;
    }
    await supabase.from('mock_scores').upsert({
      id: score.id,
      user_id: score.userId,
      subject_id: score.subjectId,
      score: score.score,
      max_score: score.maxScore,
      taken_at: score.takenAt,
      notes: score.notes ?? null,
      created_at: score.createdAt,
    });
  },

  async deleteScore(scoreId: string, mode: Mode): Promise<void> {
    if (mode === 'guest') {
      lsSet('pulse_mock_scores', lsGet<MockScore>('pulse_mock_scores').filter(s => s.id !== scoreId));
      return;
    }
    await supabase.from('mock_scores').delete().eq('id', scoreId);
  },
```

```svelte
<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import type { Subject, MockScore } from '$lib/types';

  export let subjects: Subject[];
  export let initial: MockScore | null = null;

  const dispatch = createEventDispatcher<{
    save: MockScore;
    cancel: void;
  }>();

  function todayStr(): string {
    return new Date().toISOString().split('T')[0];
  }

  let subjectId = initial?.subjectId ?? (subjects[0]?.id ?? '');
  let rawScore = initial ? String(initial.score) : '';
  let rawMax = initial ? String(initial.maxScore) : '100';
  let takenAt = initial?.takenAt ?? todayStr();
  let notes = initial?.notes ?? '';

  let scoreError = false;

  function handleSave() {
    const score = parseFloat(rawScore);
    const maxScore = parseFloat(rawMax);
    if (!subjectId || isNaN(score) || isNaN(maxScore) || maxScore <= 0 || score < 0) {
      scoreError = true;
      setTimeout(() => scoreError = false, 600);
      return;
    }
    const entry: MockScore = {
      id: initial?.id ?? crypto.randomUUID(),
      userId: initial?.userId ?? '',
      subjectId,
      score,
      maxScore,
      takenAt,
      notes: notes.trim() || undefined,
      createdAt: initial?.createdAt ?? new Date().toISOString(),
    };
    dispatch('save', entry);
  }

  function handleCancel() {
    dispatch('cancel');
  }

  function handleKeydown(e: KeyboardEvent) {
    if (e.key === 'Enter') handleSave();
    if (e.key === 'Escape') handleCancel();
  }
</script>
```

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { auth } from '$lib/stores/auth';
  import { scores, subjects, loadData, getAuthContext } from '$lib/stores/data';
  import { db } from '$lib/db';
  import ScoreForm from '$lib/components/scores/ScoreForm.svelte';
  import type { MockScore } from '$lib/types';

  onMount(() => loadData($auth));

  let adding = false;
  let editingScore: MockScore | null = null;
  let activeSubjectId = 'all';

  function formatScore(score: number, maxScore: number): string {
    const pct = maxScore > 0 ? Math.round((score / maxScore) * 100) : 0;
    return `${score} / ${maxScore}  (${pct}%)`;
  }

  function formatDate(d: string): string {
    const date = new Date(d + 'T00:00:00');
    return date.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
  }

  $: filteredScores = (activeSubjectId === 'all'
    ? [...$scores]
    : $scores.filter(s => s.subjectId === activeSubjectId)
  ).sort((a, b) => b.takenAt.localeCompare(a.takenAt));

  // For each subject that has scores, build a sorted list of pct values for the bar chart
  function trendData(subjectId: string): number[] {
    return $scores
      .filter(s => s.subjectId === subjectId)
      .sort((a, b) => a.takenAt.localeCompare(b.takenAt))
      .map(s => s.maxScore > 0 ? Math.round((s.score / s.maxScore) * 100) : 0);
  }

  function subjectName(subjectId: string): string {
    return $subjects.find(s => s.id === subjectId)?.name ?? 'Unknown';
  }

  function subjectColor(subjectId: string): string {
    return $subjects.find(s => s.id === subjectId)?.color ?? 'var(--primary)';
  }

  function subjectSymbol(subjectId: string): string {
    return $subjects.find(s => s.id === subjectId)?.symbol ?? '?';
  }

  // Subjects that the current user actually has scores for (for the tab bar)
  $: scoredSubjectIds = [...new Set($scores.map(s => s.subjectId))];

  function openAdd() {
    editingScore = null;
    adding = true;
  }

  function openEdit(score: MockScore) {
    editingScore = score;
    adding = true;
  }

  async function handleSave(e: CustomEvent<MockScore>) {
    const entry = e.detail;
    const { userId, mode } = getAuthContext();
    const withUser: MockScore = { ...entry, userId };
    scores.update(all => {
      const idx = all.findIndex(s => s.id === withUser.id);
      if (idx >= 0) { const next = [...all]; next[idx] = withUser; return next; }
      return [withUser, ...all];
    });
    await db.saveScore(withUser, mode);
    adding = false;
    editingScore = null;
  }

  function handleCancel() {
    adding = false;
    editingScore = null;
  }

  async function deleteScore(score: MockScore) {
    if (!window.confirm(`Delete this score (${formatScore(score.score, score.maxScore)}) for ${subjectName(score.subjectId)}?`)) return;
    const { mode } = getAuthContext();
    scores.update(all => all.filter(s => s.id !== score.id));
    await db.deleteScore(score.id, mode);
  }
</script>
```
