```svelte
<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import { SUBJECT_COLOR_PALETTE } from '$lib/constants';
  import ConfirmDialog from '$lib/components/ConfirmDialog.svelte';
  import type { Subject } from '$lib/types';

  export let open = false;
  export let subject: Subject | null = null;

  const dispatch = createEventDispatcher<{
    save: { name: string; symbol: string; color: string; colorRgb: string };
    delete: void;
    close: void;
  }>();

  type Swatch = { hex: string; rgb: string };

  let name = '';
  let symbol = '';
  let color: Swatch = SUBJECT_COLOR_PALETTE[0];
  let saving = false;
  let confirmOpen = false;

  // seed the form exactly once on each false→true transition of `open`
  let wasOpen = false;
  $: if (open && !wasOpen) seed();
  $: wasOpen = open;

  function seed() {
    if (subject) {
      name = subject.name;
      symbol = subject.symbol;
      color = SUBJECT_COLOR_PALETTE.find((c) => c.hex === subject.color) ?? {
        hex: subject.color,
        rgb: subject.colorRgb,
      };
    } else {
      name = '';
      symbol = '';
      color = SUBJECT_COLOR_PALETTE[0];
    }
    saving = false;
  }

  // add mode only: auto-derive a 2-char symbol from the name as the user types
  $: if (open && !subject && name) symbol = name.replace(/\s+/g, '').slice(0, 2);

  function close() {
    open = false;
    dispatch('close');
  }

  function save() {
    if (saving || !name.trim()) return;
    saving = true;
    dispatch('save', {
      name: name.trim(),
      symbol: (symbol || name.slice(0, 2)).slice(0, 2),
      color: color.hex,
      colorRgb: color.rgb,
    });
    close();
  }

  function requestDelete() {
    confirmOpen = true;
  }
  function doDelete() {
    dispatch('delete');
    close();
  }
</script>

<ConfirmDialog

  on:confirm={doDelete}
/>

{#if open}

<script lang="ts">
  import { onMount } from 'svelte';
  import { goto } from '$app/navigation';
  import { auth } from '$lib/stores/auth';
  import { subjects, topics, srsCards, dueCards, loadData, getAuthContext } from '$lib/stores/data';

  import BottomSheet from '$lib/components/BottomSheet.svelte';
  import { templatesRepo, subjectsRepo } from '$lib/data/entities';
  import { newId } from '$lib/id';
  import type { Subject, Template, SubjectPreset } from '$lib/types';

  onMount(() => loadData($auth));

  $: if ($auth.mode !== 'loading') loadData($auth);

  $: dueBySubject = (subjectId: string) => {
    const subjectTopicIds = new Set($topics.filter((t) => t.subjectId === subjectId).map((t) => t.id));
    return $dueCards.filter((c) => subjectTopicIds.has(c.topicId)).length;
  };

  // add flow (edit now lives on the subject detail page)
  let editOpen = false;

  function openAdd() {
    editOpen = true;
  }

  async function handleSave(e: CustomEvent<{ name: string; symbol: string; color: string; colorRgb: string }>) {
    const f = e.detail;
    const { userId } = getAuthContext();
    const maxOrder = $subjects.reduce((m, s) => Math.max(m, s.orderIndex), -1);
    const subject: Subject = {
      id: newId(),
      userId,
      name: f.name,
      symbol: f.symbol,
      color: f.color,
      colorRgb: f.colorRgb,
      orderIndex: maxOrder + 1,
    };
    await subjectsRepo.save(subject);
  }

  // overflow menu → save current subjects as a template
  let overflowOpen = false;

  async function saveCurrentAsTemplate() {
    const name = prompt('Template name:');
    if (!name?.trim()) return;
    const { userId } = getAuthContext();
    const presets: SubjectPreset[] = $subjects.map((sub) => ({
      name: sub.name,
      symbol: sub.symbol,
      color: sub.color,
      colorRgb: sub.colorRgb,
      topics: $topics.filter((t) => t.subjectId === sub.id).map((t) => t.name),
    }));
    const template: Template = {
      id: newId(),
      userId,
      name: name.trim(),
      presets,
      createdAt: new Date().toISOString(),
    };
    await templatesRepo.save(template);
  }
</script>
```
