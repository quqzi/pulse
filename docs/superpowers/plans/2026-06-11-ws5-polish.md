**Goal:** Land the four WS5 polish items from the roadmap spec: midnight-aware "today" derivations, toast→undo for destructive actions, an accessibility pass, and decomposition of the one remaining oversized component (WeekGrid, 756 lines).

**Architecture:** A ticking `todayDate` store replaces ad-hoc `localDateStr(new Date())` captures inside derived stores so everything recomputes when the date flips. The toast store grows an optional `action` (label + callback) that `Toaster` renders as a button — undo wires destructive repo calls to it with a pre-delete snapshot. WeekGrid's pure time-grid math moves to a tested util module. The a11y pass is audit-driven via `chrome-devtools-mcp:a11y-debugging`.

**Tech Stack:** SvelteKit 2 + TypeScript (Svelte 4 idiom), Vitest with fake timers, chrome-devtools MCP for the audit.

```ts
// src/lib/stores/clock.ts
import { readable } from 'svelte/store';
import { browser } from '$app/environment';
import { localDateStr } from '$lib/utils';

function msUntilNextMidnight(now: Date): number {
  const next = new Date(now);
  next.setHours(24, 0, 0, 0);
  return Math.max(1000, next.getTime() - now.getTime());
}

/** Local date string (YYYY-MM-DD) that ticks over at midnight and on tab re-focus. */
export const todayDate = readable(localDateStr(new Date()), (set) => {
  let timer: ReturnType<typeof setTimeout>;
  const update = () => {
    set(localDateStr(new Date()));
    timer = setTimeout(update, msUntilNextMidnight(new Date()));
  };
  timer = setTimeout(update, msUntilNextMidnight(new Date()));

  const onVisible = () => { if (!document.hidden) set(localDateStr(new Date())); };
  if (browser) document.addEventListener('visibilitychange', onVisible);
  return () => {
    clearTimeout(timer);
    if (browser) document.removeEventListener('visibilitychange', onVisible);
  };
});
```

Note: the fake-timer test relies on the readable's start function running on subscribe — subscribe inside the test before advancing timers (the test above does).

```ts
export const todaysSessions = derived([sessions, todayDate], ([$s, $today]) =>
  $s.filter(s => localDateStr(new Date(s.startedAt)) === $today)

// todayHours unchanged (derives from todaysSessions)

export const streak = derived([sessions, todayDate], ([$s, $today]) => {
  if ($s.length === 0) return 0;
  const days = new Set($s.map(s => localDateStr(new Date(s.startedAt))));
  let count = 0;
  const d = new Date($today + 'T00:00:00');
  let base = localDateStr(d);
  if (!days.has(base)) {
    d.setDate(d.getDate() - 1);
    base = localDateStr(d);
    if (!days.has(base)) return 0;
  }
  while (days.has(base)) {
    count++;
    d.setDate(d.getDate() - 1);
    base = localDateStr(d);
  }
  return count;
});

export const tasksDueToday = derived([tasks, todayDate], ([$tasks, $today]) =>
  $tasks.filter(t => !t.completedAt && t.dueDate && t.dueDate <= $today)
);

export const dueDecks = derived([topics, flashcards, todayDate], ([$topics, $flashcards, $today]) => {
  const countFor = (id: string) => $flashcards.filter(c => c.topicId === id).length;
  return $topics.filter(tp => isDeckDue(deckState(tp.nextStudyDate, countFor(tp.id), $today)));
});
```

```ts
export interface ToastAction { label: string; fn: () => void | Promise<void> }
export interface Toast {
  id: string;
  kind: ToastKind;
  message: string;
  action?: ToastAction;
}

function push(kind: ToastKind, message: string, action?: ToastAction, ttlMs = action ? 6000 : 4000): string {
  const id = `t${++seq}`;
  toasts.update((list) => [...list, { id, kind, message, action }]);
  if (ttlMs > 0) setTimeout(() => dismissToast(id), ttlMs);
  return id;
}

export function runToastAction(id: string): void {
  let act: ToastAction | undefined;
  toasts.update((list) => {
    act = list.find((t) => t.id === id)?.action;
    return list.filter((t) => t.id !== id);
  });
  void act?.fn();
}

export const toast = {
  error: (message: string, action?: ToastAction) => push('error', message, action),
  success: (message: string, action?: ToastAction) => push('success', message, action),
  info: (message: string, action?: ToastAction) => push('info', message, action),
};
```
